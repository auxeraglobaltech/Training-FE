class my_item extends uvm_sequence_item; `uvm_object_utils_begin(my_item) `uvm_field_int(addr, UVM_ALL_ON | UVM_HEX) `uvm_field_int(data, UVM_ALL_ON | UVM_HEX) `uvm_field_int(write, UVM_ALL_ON | UVM_BIN | UVM_NOCOMPARE) `uvm_object_utils_end rand logic[31:0] addr, data; rand logic write; constraint c_align { addr[1:0]==2'b00; } function new(string name="my_item"); super.new(name); endfunction function string convert2string(); return $sformatf("A=%0h D=%0h W=%0b",addr,data,write); endfunction endclass
FlagEffect
UVM_ALL_ONEnable copy, compare, print, pack
UVM_NOCOMPARESkip in scoreboard compare
UVM_NOCOPYSkip in copy/clone
UVM_NOPRINTSkip in print/sprint
UVM_HEX / UVM_BIN / UVM_DECPrint radix

Sequence · Driver · Monitor patterns

// ── SEQUENCE body ────────────────────────────────── task body(); my_item item; repeat(n) begin item = my_item::type_id::create("item"); start_item(item); if(!item.randomize() with {write==1;}) `uvm_fatal("RAND","failed") finish_item(item); end endtask // Macro shorthand: `uvm_do_with(item, { item.write==1; item.addr==32'h100; }) // ── DRIVER run_phase loop ────────────────────────── task run_phase(uvm_phase phase); my_item req; @(posedge vif.resetn); // wait for reset forever begin seq_item_port.get_next_item(req); drive_item(req); // pin-level task seq_item_port.item_done(); end endtask // ── MONITOR run_phase loop ───────────────────────── task run_phase(uvm_phase phase); my_item item; forever begin @(posedge vif.clk); if(vif.valid && vif.ready) begin item = my_item::type_id::create("item"); item.addr = vif.addr; item.data = vif.data; ap.write(item); // broadcast to subscribers end end endtask

Callbacks · Events · Barrier

// Callback — hook in component class drv_cbs extends uvm_callback; virtual task pre_drive( my_driver d, my_item i); endtask endclass // In component: `uvm_register_cb(my_driver,drv_cbs) `uvm_do_callbacks(my_driver,drv_cbs, pre_drive(this,item)) // Add in test: uvm_callbacks#(my_driver,drv_cbs):: add(m_drv, err_cb); // Events — synchronise processes uvm_event e = uvm_event_pool::get_global("done"); e.trigger(); // fire e.wait_trigger(); // block until fire // Barrier — sync N parallel procs uvm_barrier bar = new("bar"); bar.set_threshold(3); bar.wait_for(); // all 3 release together // Heartbeat — detect stuck components uvm_heartbeat hb; hb.add(m_driver); hb.start(hb_event);

// Register field access types (RW/RO/WO/W1C…) class ctrl_reg extends uvm_reg; uvm_reg_field enable, mode; function void build(); enable = uvm_reg_field:: type_id::create("enable"); // configure(parent,size,lsb,"RW", // volatile,reset,has_reset) enable.configure(this,1,0, "RW",0,1'b0,1); endfunction endclass // Register block map default_map = create_map( "map", 32'h4000_0000, 4, UVM_LITTLE_ENDIAN); default_map.add_reg(ctrl,32'h00,"RW"); lock_model(); // Test — frontdoor read/write uvm_status_e status; reg_model.ctrl.write( status, 32'h3); reg_model.ctrl.read( status, rd_data); // Field-level access reg_model.ctrl.enable.write( status, 1'b1); // Backdoor (no bus traffic) reg_model.ctrl.write( status, 32'h7, UVM_BACKDOOR); // Sync model ↔ DUT reg_model.ctrl.mirror( status, UVM_CHECK); reg_model.ctrl.update(status); // Built-in register tests: // +UVM_TESTNAME=uvm_reg_hw_reset_seq // +UVM_TESTNAME=uvm_reg_bit_bash_seq // +UVM_TESTNAME=uvm_reg_access_seq
